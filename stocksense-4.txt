     │     Guard n_trials < 1 -> raise ValueError."""                          │

     │ def deflated_sharpe_ratio(                                              │

     │ ) -> float:                                                             │

     │     1. Split rows into s even contiguous subsets (require s even, s >=  │
     │ 4).                                                                     │
     │     2. For every combination of s/2 subsets as IS                       │
     │ (itertools.combinations),                                               │
     │        the complement is OOS.                                           │
     │     3. n_star = column with the best IS mean performance.               │
     │     4. w = relative rank of n_star among all columns in OOS, in (0,1):  │
     │        w = (rank_of_n_star) / (n_cols + 1), rank ascending so 1.0 =     │
     │ best.                                                                   │
     │     5. logit = ln(w / (1 - w)).                                         │
     │     6. PBO = fraction of combinations with logit <= 0 (i.e. n_star      │
     │ landed at or                                                            │
     │        below the OOS median).                                           │
     │     Return {"pbo": float, "logits": list[float], "n_combinations":      │
     │ int}.                                                                   │
     │     PBO <= 0.5 is the bar; PBO near 1.0 means the selection is pure     │
     │ overfitting."""                                                         │
